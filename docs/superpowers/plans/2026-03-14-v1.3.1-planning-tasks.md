```python
"""Dual-mode bridge between MCP Tasks and JobManager."""

from __future__ import annotations

from datetime import datetime, timezone
from typing import Any, Callable, Dict, List, Optional

from td_mcp.capabilities import CapabilitySet
from td_mcp.jobs.manager import JobManager


class TaskAdapter:
    """Routes job lifecycle events to MCP Tasks (if supported) or job-resource polling."""

    def __init__(
        self,
        job_manager: JobManager,
        capabilities: CapabilitySet,
        notify_fn: Optional[Callable] = None,
    ):
        self._jm = job_manager
        self._caps = capabilities
        self._notify_fn = notify_fn
        self._task_meta: Dict[str, Dict[str, Any]] = {}

    @property
    def tasks_supported(self) -> bool:
        return self._caps.supports_tasks

    def wrap_job(self, job_id: str) -> Dict[str, Any]:
        job = self._jm.get_job(job_id)
        if job is None:
            raise ValueError("Job not found: {}".format(job_id))
        if not self.tasks_supported:
            return job
        meta = {
            "task_id": job_id,
            "created_at": datetime.now(timezone.utc).isoformat(),
            "status": "running",
        }
        self._task_meta[job_id] = meta
        result = dict(job)
        result["mcp_task"] = meta
        return result

    def on_progress(self, job_id: str, progress: float, message: str = "") -> None:
        self._jm.update_job(job_id, progress=progress, message=message)
        if job_id in self._task_meta:
            self._task_meta[job_id]["progress"] = progress
            self._task_meta[job_id]["message"] = message
        if self._notify_fn:
            try:
                self._notify_fn("td://job/{}".format(job_id))
            except Exception:
                pass

    def on_complete(self, job_id: str, result: Any = None) -> None:
        now = datetime.now(timezone.utc).isoformat()
        self._jm.update_job(
            job_id, status="completed", progress=1.0,
            result=result, completed_at=now,
        )
        if job_id in self._task_meta:
            self._task_meta[job_id]["status"] = "completed"
            self._task_meta[job_id]["completed_at"] = now
        if self._notify_fn:
            try:
                self._notify_fn("td://job/{}".format(job_id))
            except Exception:
                pass
```

```python
async def _with_undo_block(td_client, label, async_fn, *args):
    """Wrap an async operation in a TD undo block (start_undo_block / end_undo_block)."""
    await td_client.request("project/lifecycle",
        {"action": "start_undo_block", "label": label})
    try:
        result = await async_fn(*args)
        return result
    finally:
        await td_client.request("project/lifecycle",
            {"action": "end_undo_block"})
```

```python
@mcp.tool(name="td_plan_patch")
async def td_plan_patch(params: PlanPatchInput, ctx: Context) -> Dict[str, Any]:
    """Generate a structured patch plan for an intent. Inspects current state, validates
    op types against knowledge cards. Does NOT mutate the project."""
    finish = _start_tool(ctx, "td_plan_patch")
    try:
        client = _get_client(ctx)
        idx = _get_card_index(ctx)

        # Inspect target path children
        try:
            children = await client.request("get_nodes", {"path": params.target_path})
        except Exception:
            children = {}

        existing_nodes = children.get("nodes", []) if isinstance(children, dict) else []

        # Parse intent for op type references (simple heuristic)
        steps = []
        risks = []
        validation_checks = ["td_get_errors after execution"]
        dependencies = []

        # If recipe_id provided, load recipe for structure
        recipe_ops = []
        if params.recipe_id:
            store = _get_technique_store(ctx)
            entry = store.get(params.recipe_id, scope="project") or store.get(params.recipe_id, scope="global")
            if entry:
                technique = entry.get("technique", {})
                recipe = technique.get("recipe", {})
                recipe_nodes = recipe.get("nodes", {})
                for rel_path, node_info in recipe_nodes.items():
                    if isinstance(node_info, dict) and node_info.get("type"):
                        op_type = node_info["type"]
                        recipe_ops.append(op_type)
                        card = idx.get_operator(op_type) if idx else None
                        step = {
                            "action": "create",
                            "op_type": op_type,
                            "path": "{}/{}".format(params.target_path.rstrip("/"), rel_path.lstrip("/")),
                            "card_found": card is not None,
                        }
                        steps.append(step)

        plan = {
            "intent": params.intent,
            "target_path": params.target_path,
            "existing_children": len(existing_nodes),
            "steps": steps,
            "dependencies": dependencies,
            "risks": risks,
            "validation_checks": validation_checks,
            "recipe_id": params.recipe_id,
        }

        _audit_log(ctx, "td_plan_patch", {"intent": params.intent, "target": params.target_path})
        return plan
    except Exception as exc:
        _record_tool_error(ctx, "td_plan_patch")
        return {"error": str(exc)}
    finally:
        finish()
```

```python
@mcp.tool(name="td_validate_recipe")
async def td_validate_recipe(params: ValidateRecipeInput, ctx: Context) -> Dict[str, Any]:
    """Validate a technique recipe: check required op types exist, params are valid,
    build compatibility."""
    finish = _start_tool(ctx, "td_validate_recipe")
    try:
        idx = _get_card_index(ctx)
        svc = _get_services(ctx)

        recipe = params.recipe
        if recipe is None and params.recipe_id:
            store = _get_technique_store(ctx)
            entry = store.get(params.recipe_id, scope=params.scope)
            if not entry:
                return {"valid": False, "error": "Recipe not found: {}".format(params.recipe_id)}
            technique = entry.get("technique", {})
            recipe = technique.get("recipe")

        if not recipe:
            return {"valid": False, "error": "No recipe data provided or found"}

        missing_ops = []
        invalid_params = []
        build_warnings = []

        nodes = recipe.get("nodes", {})
        for rel_path, node_info in nodes.items():
            if not isinstance(node_info, dict):
                continue
            op_type = node_info.get("type", "")
            if op_type and idx:
                card = idx.get_operator(op_type)
                if card is None:
                    missing_ops.append(op_type)
                elif card and svc.td_build:
                    compat = idx.check_compatibility(op_type, svc.td_build)
                    if compat.get("status") == "incompatible":
                        build_warnings.append({
                            "op_type": op_type,
                            "reason": compat.get("reason", ""),
                        })

        valid = len(missing_ops) == 0 and len(invalid_params) == 0
        return {
            "valid": valid,
            "missing_ops": missing_ops,
            "invalid_params": invalid_params,
            "build_compatible": len(build_warnings) == 0,
            "build_warnings": build_warnings,
            "node_count": len(nodes),
        }
    except Exception as exc:
        _record_tool_error(ctx, "td_validate_recipe")
        return {"error": str(exc)}
    finally:
        finish()
```

```python
@mcp.tool(name="td_audit_project")
async def td_audit_project(params: AuditProjectInput, ctx: Context) -> Dict[str, Any]:
    """Audit a project subtree: structure summary, palette usage, custom COMPs,
    fragile zones, and build warnings."""
    finish = _start_tool(ctx, "td_audit_project")
    try:
        client = _get_client(ctx)
        idx = _get_card_index(ctx)
        svc = _get_services(ctx)

        # Get all nodes in subtree
        try:
            tree = await client.request("get_nodes", {"path": params.root_path, "recursive": True})
        except Exception as exc:
            return {"error": "Cannot read project tree: {}".format(exc)}

        nodes = tree.get("nodes", []) if isinstance(tree, dict) else []

        family_counts = {}
        op_type_counts = {}
        palette_components = []
        custom_comps = []
        error_nodes = []

        for node in nodes:
            if not isinstance(node, dict):
                continue
            family = node.get("family", "unknown")
            family_counts[family] = family_counts.get(family, 0) + 1
            op_type = node.get("type", "")
            if op_type:
                op_type_counts[op_type] = op_type_counts.get(op_type, 0) + 1
            # Detect palette components
            if idx and op_type:
                palette_card = idx.get_palette(op_type)
                if palette_card:
                    palette_components.append(node.get("path", op_type))
            # Detect COMPs
            if family == "COMP" and not node.get("isBase", False):
                custom_comps.append(node.get("path", ""))

        # Check for errors
        try:
            errors = await client.request("get_errors", {"path": params.root_path, "recursive": True})
            if isinstance(errors, dict):
                error_nodes = errors.get("errors", [])
        except Exception:
            pass

        build_warnings = []
        if idx and svc.td_build:
            for op_type in op_type_counts:
                compat = idx.check_compatibility(op_type, svc.td_build)
                if compat.get("status") in ("caution", "incompatible"):
                    build_warnings.append({"op_type": op_type, "status": compat["status"]})

        return {
            "root_path": params.root_path,
            "total_nodes": len(nodes),
            "family_counts": family_counts,
            "op_type_counts": op_type_counts,
            "palette_components": palette_components,
            "custom_comps": custom_comps[:20],  # Limit output size
            "error_count": len(error_nodes) if isinstance(error_nodes, list) else 0,
            "build_warnings": build_warnings,
            "td_build": svc.td_build,
        }
    except Exception as exc:
        _record_tool_error(ctx, "td_audit_project")
        return {"error": str(exc)}
    finally:
        finish()
```
